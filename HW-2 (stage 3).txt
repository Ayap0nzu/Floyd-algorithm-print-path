import copy

n = int(input())

fw_array = [[0 for j in range(n)] for i in range(n)]

def find_shortest_path(start, end, R):
    shortest_path = []

    shortest_path.append(start)
    if start == end:
        shortest_path.append(end)
        return shortest_path

for i in range(n):
    value = input().split()
    for j in range(n):
        if value[j] == '*':
            fw_array[i][j] = float('inf')
        else:
            fw_array[i][j] = int(value[j])
        

for k in range(n):
    for i in range(n):
        for j in range(n):
            fw_array[i][j] = min(fw_array[i][j],
                                 fw_array[i][k] + fw_array[k][j]
                                 )

for i in range(n):
    for j in range(n):
        distance = fw_array[i][j]
        print(f"({i+1},{j+1}) : distance = {distance}, path =", find_shortest_path(i+1, j+1, fw_array))
